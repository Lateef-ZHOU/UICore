cmake_minimum_required(VERSION 3.28)
project(LidarUICore LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11.2 REQUIRED COMPONENTS Quick Qml)
file(GLOB_RECURSE views CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "qml/*.qml")
foreach(view IN LISTS views)
    get_filename_component(name "${view}" NAME)
    set_source_files_properties("${view}" PROPERTIES QT_RESOURCE_ALIAS "${name}")
endforeach()
set_source_files_properties(qml/Controls/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(LidarUICore STATIC URI Lidar.UI VERSION 1.0
    CLASS_NAME LidarUIPlugin RESOURCE_PREFIX /qt/qml
    OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/Lidar/UI" QML_FILES ${views})
add_library(Lidar::UICore ALIAS LidarUICore)
target_compile_features(LidarUICore PUBLIC cxx_std_20)
target_link_libraries(LidarUICore PUBLIC Qt6::Quick)
target_compile_definitions(LidarUICore PUBLIC QT_NO_KEYWORDS)

if(MSVC)
    target_compile_options(LidarUICore PRIVATE /W4)
    get_target_property(module_sources LidarUICore SOURCES)
    foreach(source IN LISTS module_sources)
        if(source MATCHES "/[.]rcc/qmlcache/")
            set_source_files_properties("${source}" PROPERTIES COMPILE_OPTIONS "/wd4702")
        endif()
    endforeach()
endif()

# 作为依赖时不接管宿主的测试开关。
if(PROJECT_IS_TOP_LEVEL)
    include(CTest)
    if(BUILD_TESTING)
        add_subdirectory(tests)
    endif()
endif()
